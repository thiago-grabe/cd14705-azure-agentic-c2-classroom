# so the install succeeds and then fails at import time with
# `ModuleNotFoundError: No module named 'httpx'`.
#
# requirements.txt is left exactly as the course ships it; install with:
#   uv pip install -r requirements.txt -c constraints.txt
openai<3
httpx>=0.27,<0.29
